\begin{frame}{Motivation}
\begin{itemize}
    \item\kw{Improved Parking Experience}
    \begin{itemize}
        \item Pay online without physical tickets that need management and can be lost
        \item License plate-based identification
        \item No need to interact with payment terminals
    \end{itemize}

    \item\kw{Smart Booking System}
    \begin{itemize}
        \item Reserve parking spots in advance
        \item Find closest parking facilities with available spaces
        \item Extension: Book spots periodically (daily, weekly, monthly)
    \end{itemize}
        
\end{itemize}
\end{frame}

\begin{itemize}
    \item\kw{Dataset: Car License Plate Detection (Kaggle)}
    \begin{itemize}
        \item 400+ annotated images (front/rear views)
        \item Suitable for prototyping and evaluation
    \end{itemize}

    \item\kw{Computer Vision: YOLOv11+OCR}
    \begin{itemize}
        \item Real-time license plate detection on the edge
        \item \textbf{OCR} extracts the license plate string
    \end{itemize}

    \item\kw{Messaging: NATS}
    \begin{itemize}
        \item Pub/Sub for camera frames and control signals (barrier)
        \item Asynchronous cloud-to-edge communication
    \end{itemize}    
\end{itemize}

\begin{column}{0.5\textwidth}
\begin{itemize}
    \item\kw{Camera Implementation}
    \begin{itemize}
        \item Python process simulating a camera at the entry/exit gate
        \item Sends image stream to edge server over NATS
        \item Car display duration 1 second
        \item Entry switches between car and empty gate scenes every 4 seconds
        \item Exit shows entered car exiting randomly
    \end{itemize}

    \begin{itemize}

        \item \textbf{SQLite} as lightweight local store
        \item \textbf{FastAPI} to monitor barrier state and camera input
        \item \textbf{Dockerized} components for easy deployment on edge devices with docker-compose
    \end{itemize}

    \item<+-> \kw{Cloud Stack: Akka Typed + Akka HTTP}
    \begin{itemize}
        \item Actor-based orchestration for \textit{parking lot} functionality
        \item \textbf{REST API} for edge and web clients
    \end{itemize}
\end{itemize}

          \item Separate instances for:
            \begin{itemize}
              \item NATS server
              \item Akka backend
              \item Nginx webserver
            \end{itemize}
      \end{itemize}

  \item<+-> \kw{Edge Client simulation for load testing}
      \begin{itemize}
        \item wrk2 HTTP benchmarking tool simulating client requests to backend
        \item Client simulation host with Intel i5-8600K CPU, 16 GiB RAM   
      \end{itemize}

  \item<+-> \kw{Edge setup for latency measurements}
      \begin{itemize}
        \item Host with Intel i5-8250U CPU, 8 GiB RAM, no GPU   
      \end{itemize}
\end{itemize}
